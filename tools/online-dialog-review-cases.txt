test('REPRO: selecting a legal fight target hides its legal battle choice, while the all-actions path still resolves it',async()=>{
 const h=harness();await h.login();await h.click('online-create');await h.friend();
 await h.editGame(s=>{
  s.phase='main';s.active=0;s.decision=null;
  const c=(uid,cardId)=>({uid,cardId,entered:0,tapped:false});
  s.players[0].battle=[];s.players[0].hand=[c('fighter-review','dmr21-092')];
  s.players[0].mana=Array.from({length:15},(_,i)=>c('mana-review-'+i,i%2?'dm01-027':'dm01-097'));
  s.players[1].battle=[c('fight-target-review','dm01-108')];
  e.step(s,e.getLegalActions(s).find(a=>a.kind==='play'&&a.uid==='fighter-review'));
 });
 assert.equal(h.game().decision.type,'fight');
 const legal=h.read('onlineState.room.game.actions.find(a=>a.kind==="fight")');assert.ok(legal);
 assert.ok(!legal.targetUids?.includes('fight-target-review'));
 const before=h.game().actions;
 await h.click('online-card',{uid:'fight-target-review'});
 assert.match(h.root.innerHTML,/このカードで選べる操作はありません/);
 assert.ok(!h.root.innerHTML.includes('data-action="online-move"'));
 assert.equal(h.game().actions,before);
 await h.click('online-all-actions');
 await h.click('online-move',{index:String(legal.id),revision:String(h.read('onlineState.room.revision'))});
 assert.ok(!h.game().players[1].battle.some(c=>c.uid==='fight-target-review'));
 console.log('CONFIRMED legal target is omitted from targetUids; tapping it reports no action, but all-actions can execute the fight');
 h.read('onlineClient.dispose()');
});
